package de.jplag.java_cpg.passes;

import java.util.ArrayList;

import org.jetbrains.annotations.NotNull;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import de.fraunhofer.aisec.cpg.TranslationContext;
import de.fraunhofer.aisec.cpg.TranslationResult;
import de.fraunhofer.aisec.cpg.passes.TranslationResultPass;
import de.fraunhofer.aisec.cpg.passes.configuration.DependsOn;
import de.jplag.Token;
import de.jplag.java_cpg.token.CpgTokenizer;
import de.jplag.java_cpg.visitor.NodeOrderStrategy;

/**
 * This pass tokenizes the {@link de.fraunhofer.aisec.cpg.TranslationResult}. It is a noncompiled Java reference counterpart
 * of {@link TokenizationPass}, originally introduced as a workaround to make Javadoc work. If the package 'passes'
 * contains only Kotlin files, Javadoc fails.
 */
@DependsOn(CpgTransformationPass.class)
public class JTokenizationPass extends TranslationResultPass {

    private static final Logger logger = LoggerFactory.getLogger(JTokenizationPass.class);
    private final ArrayList<Token> tokenList = new ArrayList<>();
    private final NodeOrderStrategy strategy = new NodeOrderStrategy();

    /**
     * Constructs a new JTokenizationPass.
     * @param ctx the current {@link de.fraunhofer.aisec.cpg.TranslationContext}
     */
    public JTokenizationPass(@NotNull TranslationContext ctx) {
        super(ctx);
    }

    @Override
    public void accept(TranslationResult translationResult) {
        tokenList.clear();

        var tokenizer = new CpgTokenizer();

        strategy.walk(translationResult, node -> tokenizer.visit(node).ifPresent(tokenList::add),
                node -> tokenizer.exit(node).ifPresent(tokenList::add));
        translationResult.getScratch().put("tokenList", tokenList);
    }

    @Override
    public void cleanup() {
        logger.info("Found {} tokens", tokenList.size());
    }

}
